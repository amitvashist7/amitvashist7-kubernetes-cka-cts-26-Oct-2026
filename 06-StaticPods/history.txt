## Create a Static POD
kubectl run hello-static-k9s --image=nginx --port=80 --dry-run -o yaml
kubectl get nodes 
docker exec -it <worker-node-name> bash
cat /etc/kubernetes/kubelet.conf
ls
cat /var/lib/kubelet/config.yaml
cd /etc/kubernetes/manifests/
echo "<content of yaml>" > /etc/kubernetes/manifests/nginx.yaml
exit 
kubectl get pods 
kubectl delete pods --all 

## Now Delete the Static POD
docker exec -it <worker-node-name> bash
rm -rf /etc/kubernetes/manifests/nginx.yaml
exit
